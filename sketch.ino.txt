#include <WiFi.h>
#include <HTTPClient.h>

const char* ssid = "Wokwi-GUEST";
const char* password = "";
String apiKey = "YOUR_WRITE_API_KEY";   // <-- paste your ThingSpeak Write API Key here
const char* server = "http://api.thingspeak.com";

int pirPin = 4;   // PIR D pin connected here
int ledPin = 2;   // LED connected here

void setup() {
  Serial.begin(115200);
  pinMode(pirPin, INPUT);
  pinMode(ledPin, OUTPUT);

  WiFi.begin(ssid, password);
  Serial.print("Connecting to WiFi");
  while (WiFi.status() != WL_CONNECTED) {
    delay(500);
    Serial.print(".");
  }
  Serial.println("\nConnected!");
}

void loop() {
  int motion = digitalRead(pirPin);
  digitalWrite(ledPin, motion);

  if (WiFi.status() == WL_CONNECTED) {
    HTTPClient http;
    String url = String(server) + "/update?api_key=" + apiKey + "&field1=" + String(motion);
    http.begin(url);
    int httpCode = http.GET();
    Serial.print("Upload status: ");
    Serial.println(httpCode);
    http.end();
  }

  delay(15000);   // ThingSpeak free tier needs min 15 sec gap
}
